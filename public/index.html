function copySnippet() {
  navigator.clipboard.writeText(snippetTag()).then(() => {
    const el = document.getElementById('snippet-copied');
    el.textContent = 'Copied!';

async function createExperiment() {
  const name = document.getElementById('exp-name').value;
  const url_match = document.getElementById('exp-url').value;
  const el = document.getElementById('exp-result');
  const result = await callApi('/api/experiments', {
    method: 'POST', headers: authHeaders(), body: JSON.stringify({ name, url_match })
  });
  el.className = result.ok ? 'ok-msg' : 'error-msg';
  if (result.ok) {
    el.textContent = `Created. Experiment ID: ${result.data.id}. Make sure the snippet above is installed on the target page.`;
    // Auto-fill downstream sections so the id doesn't need copy-pasting around.
    ['var-exp-id', 'manage-exp-id', 'status-exp-id', 'results-exp-id'].forEach((fieldId) => {
      document.getElementById(fieldId).value = result.data.id;
    });
    // If the url_match looks like a real URL (not just a path fragment), pre-fill the preview URL too.
    if (/^https?:\/\//.test(url_match)) document.getElementById('manage-page-url').value = url_match;
    loadExperimentList();
  } else {
    el.textContent = `Error: ${result.message}`;
  }
}

// --- Change / Goal row builder (replaces hand-written JSON) ---

const CHANGE_TYPES = {
  text:  { label: 'Change text',        valueLabel: 'New text',              placeholder: 'Get Started Free', multiline: false },
  html:  { label: 'Replace HTML',       valueLabel: 'New HTML',              placeholder: 'New content', multiline: true },
  hide:  { label: 'Hide element',       valueLabel: null },
  show:  { label: 'Show element',       valueLabel: null },
  css:   { label: 'Custom CSS',         valueLabel: 'CSS declarations',      placeholder: 'background: #1AB7C8; padding: 12px; border-radius: 6px;', multiline: true },
  attr:  { label: 'Change attribute',   valueLabel: 'attr=value (e.g. href=/checkout-v2)', placeholder: 'href=/checkout-v2', multiline: false },
  js:    { label: 'Custom JS (advanced)', valueLabel: 'JS to run — "el" is the matched element', placeholder: 'el.addEventListener("click", () => console.log("clicked"));', multiline: true },
};

function addChangeRow(prefill) {
  const list = document.getElementById('changes-list');
  const row = document.createElement('div');
  row.className = 'change-row';
  const typeOptions = Object.entries(CHANGE_TYPES).map(([k, v]) => ``).join('');
  row.innerHTML = `
    

      
      
      
      
    

    

      
      
    

    

      
    

  `;
  list.appendChild(row);

  const selectEl = row.querySelector('.c-type');
  const valueWrap = row.querySelector('.c-value-wrap');
  function renderValueField() {
    const cfg = CHANGE_TYPES[selectEl.value];
    if (!cfg.valueLabel) { valueWrap.style.display = 'none'; return; }
    valueWrap.style.display = '';
    const tag = cfg.multiline ? 'textarea' : 'input';
    valueWrap.innerHTML = `<${tag} class="c-value" placeholder="${cfg.placeholder || ''}">`;
  }
  selectEl.addEventListener('change', renderValueField);
  renderValueField();

  if (prefill) {
    row.querySelector('.c-selector').value = prefill.selector || '';
    selectEl.value = prefill.type || 'text';
    renderValueField();
    if (prefill.value !== undefined) row.querySelector('.c-value').value = typeof prefill.value === 'string' ? prefill.value : JSON.stringify(prefill.value);
  }
}

function addGoalRow(prefill) {
  const list = document.getElementById('goals-list');
  const row = document.createElement('div');
  row.className = 'goal-row';
  row.innerHTML = `
    

      
      
      
      
    

    

    
  `;
  list.appendChild(row);

  const typeEl = row.querySelector('.g-type');
  const valueLabelEl = row.querySelector('.g-value-label');
  const valueEl = row.querySelector('.g-value');
  function renderValueField() {
    if (typeEl.value === 'url') {
      valueLabelEl.textContent = 'URL contains';
      valueEl.placeholder = '/thank-you';
    } else {
      valueLabelEl.textContent = 'Selector';
      valueEl.placeholder = '#cta-button';
    }
  }
  typeEl.addEventListener('change', renderValueField);

  if (prefill) {
    typeEl.value = prefill.type || 'click';
    row.querySelector('.g-id').value = prefill.id || '';
    valueEl.value = prefill.type === 'url' ? (prefill.url_match || '') : (prefill.selector || '');
  }
  renderValueField();
}

// Parses "attr=value" into { attr, value } for the attribute change type.
function parseAttrValue(str) {
  const idx = str.indexOf('=');
  if (idx === -1) return { attr: str.trim(), value: '' };
  return { attr: str.slice(0, idx).trim(), value: str.slice(idx + 1).trim() };
}

function collectChanges() {
  return Array.from(document.querySelectorAll('#changes-list .change-row')).map((row) => {
    const selector = row.querySelector('.c-selector').value.trim();
    const type = row.querySelector('.c-type').value;
    const valueField = row.querySelector('.c-value');
    const raw = valueField ? valueField.value : '';

    if (type === 'hide' || type === 'show') return { selector, type };
    if (type === 'attr') {
      const { attr, value } = parseAttrValue(raw);
      return { selector, type, attr, value };
    }
    if (type === 'css') {
      // Turn "background: red; padding: 10px;" into the {property: value} object the snippet applies via el.style.
      const style = {};
      raw.split(';').forEach((decl) => {
        const [prop, val] = decl.split(':');
        if (prop && val) {
          const camel = prop.trim().replace(/-([a-z])/g, (_, c) => c.toUpperCase());
          style[camel] = val.trim();
        }
      });
      return { selector, type: 'style', value: style };
    }
    return { selector, type, value: raw };
  }).filter((c) => c.selector);
}

function collectGoals() {
  return Array.from(document.querySelectorAll('#goals-list .goal-row')).map((row) => {
    const type = row.querySelector('.g-type').value;
    const value = row.querySelector('.g-value').value.trim();
    const id = row.querySelector('.g-id').value.trim();
    return type === 'url' ? { type: 'url', url_match: value, id } : { type: 'click', selector: value, id };
  }).filter((g) => (g.type === 'url' ? g.url_match : g.selector));
}

async function createVariant() {
  const expId = document.getElementById('var-exp-id').value.trim();
  const el = document.getElementById('var-result');
  if (!expId) { el.className = 'error-msg'; el.textContent = 'Enter an Experiment ID first (or create one above).'; return; }

  const name = document.getElementById('var-name').value;
  const traffic_split = parseInt(document.getElementById('var-split').value, 10);
  const changes = collectChanges();
  const goals = collectGoals();

  const result = await callApi(`/api/experiments/${expId}/variants`, {
    method: 'POST', headers: authHeaders(), body: JSON.stringify({ name, traffic_split, changes, goals })
  });
  el.className = result.ok ? 'ok-msg' : 'error-msg';
  el.textContent = result.ok ? `Variant created. ID: ${result.data.id}` : `Error: ${result.message}`;
}

// --- Preview & manage variants ---

function buildPreviewUrl(pageUrl, variantId) {
  const sep = pageUrl.includes('?') ? '&' : '?';
  return `${pageUrl}${sep}ab_preview=${variantId}`;
}

async function loadManageVariants() {
  const expId = document.getElementById('manage-exp-id').value.trim();
  const msgEl = document.getElementById('manage-msg');
  const listEl = document.getElementById('manage-variants-list');
  listEl.innerHTML = '';
  if (!expId) { msgEl.className = 'error-msg'; msgEl.textContent = 'Enter an Experiment ID first (or create one above).'; return; }

  const result = await callApi(`/api/experiments/${expId}`, { headers: authHeaders() });
  if (!result.ok) { msgEl.className = 'error-msg'; msgEl.textContent = `Error: ${result.message}`; return; }

  const pageUrl = document.getElementById('manage-page-url').value.trim();
  const variants = result.data.variants || [];
  if (variants.length === 0) { msgEl.className = 'hint'; msgEl.textContent = 'No variants yet on this experiment.'; return; }

  msgEl.textContent = '';
  variants.forEach((v) => {
    const card = document.createElement('div');
    card.className = 'variant-card' + (v.enabled ? '' : ' disabled');
    const previewBtn = pageUrl
      ? ``
      : ``;
    card.innerHTML = `
      

        ${v.name}
        ${v.enabled ? 'live' : 'paused'}
        
${v.traffic_split}% traffic · ${(v.changes || []).length} change(s) · ${(v.goals || []).length} goal(s)

      

      

        ${previewBtn}
        
        
      

    `;
    listEl.appendChild(card);
  });
}

async function toggleVariantEnabled(expId, variantId, enabled, btn) {
  btn.disabled = true;
  const result = await callApi(`/api/experiments/${expId}/variants/${variantId}/enabled`, {
    method: 'PATCH', headers: authHeaders(), body: JSON.stringify({ enabled })
  });
  btn.disabled = false;
  if (!result.ok) { alert(`Error: ${result.message}`); return; }
  loadManageVariants();
}

async function deleteVariant(expId, variantId, btn) {
  if (!confirm('Delete this variant permanently? This also deletes its recorded views/conversions and cannot be undone.\n\nIf you just want to stop it running while keeping its history, use Pause instead.')) return;
  btn.disabled = true;
  const result = await callApi(`/api/experiments/${expId}/variants/${variantId}`, {
    method: 'DELETE', headers: authHeaders()
  });
  btn.disabled = false;
  if (!result.ok) { alert(`Error: ${result.message}`); return; }
  loadManageVariants();
}

let rateChart, trendChart;

async function loadResults() {
  const expId = document.getElementById('results-exp-id').value.trim();
  const msgEl = document.getElementById('results-msg');
  if (!expId) { msgEl.className = 'error-msg'; msgEl.textContent = 'Enter an Experiment ID first (or create one above).'; return; }

  const result = await callApi(`/api/results/${expId}`, { headers: authHeaders() });

  if (!result.ok) {
    msgEl.className = 'error-msg';
    msgEl.textContent = `Error: ${result.message}`;
    document.getElementById('results-table').style.display = 'none';
    document.getElementById('rate-chart').style.display = 'none';
    document.getElementById('trend-chart').style.display = 'none';
    return;
  }
  msgEl.textContent = '';
  const results = result.data.results || [];
  const tbody = document.getElementById('results-body');
  tbody.innerHTML = '';
  results.forEach(r => {
    const tr = document.createElement('tr');
    tr.innerHTML = `${r.variant_name}${r.visitors}${r.clicks}${r.conversions}${r.conversion_rate}%`;
    tbody.appendChild(tr);
  });
  document.getElementById('results-table').style.display = 'table';

  // Bar chart: conversion rate per variant — the headline "which one's winning" view.
  const rateCanvas = document.getElementById('rate-chart');
  rateCanvas.style.display = results.length ? 'block' : 'none';
  if (rateChart) rateChart.destroy();
  if (results.length) {
    rateChart = new Chart(rateCanvas, {
      type: 'bar',
      data: {
        labels: results.map(r => r.variant_name),
        datasets: [{ label: 'Conversion rate (%)', data: results.map(r => r.conversion_rate), backgroundColor: '#1AB7C8' }]
      },
      options: { responsive: true, plugins: { title: { display: true, text: 'Conversion rate by variant' } }, scales: { y: { beginAtZero: true } } }
    });
  }

  // Line chart: cumulative visitors over time per variant — the trend view.
  const trendResult = await callApi(`/api/results/${expId}/timeseries`, { headers: authHeaders() });
  const trendCanvas = document.getElementById('trend-chart');
  if (trendChart) trendChart.destroy();
  if (trendResult.ok && trendResult.data.series && trendResult.data.series.length) {
    const series = trendResult.data.series;
    const days = [...new Set(series.map(r => r.day))].sort();
    const variantNames = [...new Set(series.map(r => r.variant_name))];
    const palette = ['#1AB7C8', '#1B2A2E', '#e07a5f', '#8a5cf6', '#1a8a5c'];

    const datasets = variantNames.map((name, i) => {
      let running = 0;
      const dataByDay = days.map((day) => {
        const row = series.find(r => r.day === day && r.variant_name === name);
        running += row ? row.visitors : 0;
        return running;
      });
      return { label: name, data: dataByDay, borderColor: palette[i % palette.length], backgroundColor: palette[i % palette.length], tension: 0.2, fill: false };
    });

    trendCanvas.style.display = 'block';
    trendChart = new Chart(trendCanvas, {
      type: 'line',
      data: { labels: days.map(d => new Date(d).toLocaleDateString()), datasets },
      options: { responsive: true, plugins: { title: { display: true, text: 'Cumulative visitors over time' } }, scales: { y: { beginAtZero: true } } }
    });
  } else {
    trendCanvas.style.display = 'none';
  }
}
